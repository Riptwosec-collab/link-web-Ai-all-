        // Paste Handler: วาง URL -> ดึงข้อมูล -> แยกหมวด -> สร้าง Card
        function setupHomePasteHandler() {
            const input = document.getElementById('main-url-input');
            const grid = document.getElementById('all-links-grid');

            if (!input || !grid || input.dataset.bound === '1') return;
            input.dataset.bound = '1';

            input.addEventListener('paste', async (event) => {
                let pastedText = (event.clipboardData || window.clipboardData)
                    .getData('text')
                    .trim();

                let parsedUrl;
                try {
                    parsedUrl = new URL(pastedText);
                    if (!['http:', 'https:'].includes(parsedUrl.protocol)) return;
                } catch {
                    return;
                }

                event.preventDefault();
                input.value = 'Fetching preview, image & description...';
                input.classList.add('border-brand-purple');
                input.disabled = true;

                const categorizedData = simulateCategorization(pastedText);
                const realWebData = await fetchWebData(pastedText);

                let finalData = {
                    ...(categorizedData || {}),
                    url: pastedText
                };

                if (realWebData) {
                    finalData = {
                        ...finalData,
                        title: realWebData.title || finalData.title,
                        description: realWebData.description || '',
                        imageUrl: realWebData.imageUrl || '',
                        favicon: realWebData.favicon || '',
                        domain: realWebData.domain || finalData.domain
                    };
                }

                finalData.description = compactDescription(
                    finalData.description ||
                    getFallbackDescription(finalData.domain, finalData.tags || [])
                );

                if (finalData) {
                    const cardHtml = createCardHtml(finalData);
                    grid.insertAdjacentHTML('afterbegin', cardHtml);
                }

                input.disabled = false;
                input.value = '';
                input.classList.remove('border-brand-purple');
                input.placeholder = realWebData
                    ? 'Saved with preview! Paste another URL...'
                    : 'Saved! Preview unavailable, fallback info used.';

                setTimeout(() => {
                    input.placeholder = 'Paste URL here for Auto Save & Categorize...';
                }, 3000);

                input.focus();
            });
        }

        // Small interactive touches for dedicated pages
        function setupSearchPage() {
            const input = document.getElementById('smart-search-input');
            if (!input) return;
            const items = [...document.querySelectorAll('.search-result')];
            input.addEventListener('input', () => {
                const q = input.value.trim().toLowerCase();
                items.forEach(item => {
                    const hit = item.dataset.search.toLowerCase().includes(q);
                    item.style.display = hit ? '' : 'none';
                });
            });
            input.focus();
        }

                });
            });
        }
