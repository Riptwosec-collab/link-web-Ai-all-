        function escapeHtml(value = '') {
            return String(value)
                .replaceAll('&', '&amp;')
                .replaceAll('<', '&lt;')
                .replaceAll('>', '&gt;')
                .replaceAll('"', '&quot;')
                .replaceAll("'", '&#039;');
        }

        function compactDescription(value = '', maxLength = 150) {
            const clean = String(value).replace(/\s+/g, ' ').trim();
            if (!clean) return '';
            if (clean.length <= maxLength) return clean;

            const sliced = clean.slice(0, maxLength);
            const withoutBrokenWord = sliced.replace(/\s+\S*$/, '').trim();
            return `${withoutBrokenWord || sliced.trim()}…`;
        }

        function resolveAssetUrl(assetUrl, pageUrl) {
            if (!assetUrl) return '';
            try {
                return new URL(assetUrl, pageUrl).href;
            } catch {
                return '';
            }
        }

        function getFallbackDescription(domain, tags = []) {
            const lower = String(domain || '').toLowerCase();

            if (lower.includes('openai.com') || lower.includes('chatgpt.com')) {
                return 'AI assistant and tools for writing, research, coding and everyday tasks.';
            }
            if (lower.includes('github.com')) {
                return 'Code hosting and collaboration platform for software projects and developers.';
            }
            if (lower.includes('google.com')) {
                return 'Search engine and online tools for finding information across the web.';
            }
            if (lower.includes('youtube.com')) {
                return 'Video platform for watching, sharing and discovering online content.';
            }
            if (lower.includes('netflix.com')) {
                return 'Streaming platform for movies, series and entertainment.';
            }
            if (tags.includes('AI')) {
                return 'AI-related website or online tool.';
            }
            return `Website and online service from ${domain || 'this domain'}.`;
        }

        function readMeta(doc, selectors = []) {
            for (const selector of selectors) {
                const el = doc.querySelector(selector);
                const value = el?.getAttribute('content') || el?.getAttribute('href') || el?.textContent;
                if (value && value.trim()) return value.trim();
            }
            return '';
        }

        function metadataFromHtml(htmlText, url) {
            const parser = new DOMParser();
            const doc = parser.parseFromString(htmlText, 'text/html');
            const domain = new URL(url).hostname.replace(/^www\./, '');

            const title = readMeta(doc, [
                'meta[property="og:title"]',
                'meta[name="twitter:title"]',
                'title',
                'h1'
            ]);

            const description = readMeta(doc, [
                'meta[property="og:description"]',
                'meta[name="twitter:description"]',
                'meta[name="description"]'
            ]);

            const imageRaw = readMeta(doc, [
                'meta[property="og:image:secure_url"]',
                'meta[property="og:image"]',
                'meta[name="twitter:image"]',
                'meta[name="twitter:image:src"]',
                'meta[itemprop="image"]',
                'link[rel="image_src"]'
            ]);

            const faviconRaw = readMeta(doc, [
                'link[rel="apple-touch-icon"]',
                'link[rel="icon"]',
                'link[rel="shortcut icon"]'
            ]);

            return {
                title,
                description: compactDescription(description),
                imageUrl: resolveAssetUrl(imageRaw, url),
                favicon: resolveAssetUrl(faviconRaw, url) ||
                    `https://www.google.com/s2/favicons?sz=64&domain=${encodeURIComponent(domain)}`,
                domain
            };
        }

        async function fetchFromAllOrigins(url) {
            const proxyUrl = `https://api.allorigins.win/raw?url=${encodeURIComponent(url)}`;
            const response = await fetch(proxyUrl, { signal: AbortSignal.timeout(9000) });
            if (!response.ok) throw new Error(`AllOrigins ${response.status}`);
            const htmlText = await response.text();
            return metadataFromHtml(htmlText, url);
        }

        async function fetchFromMicrolink(url) {
            const endpoint = `https://api.microlink.io/?url=${encodeURIComponent(url)}`;
            const response = await fetch(endpoint, { signal: AbortSignal.timeout(12000) });
            if (!response.ok) throw new Error(`Microlink ${response.status}`);

            const result = await response.json();
            if (result.status !== 'success' || !result.data) {
                throw new Error('Microlink returned no metadata');
            }

            const d = result.data;
            const domain = new URL(url).hostname.replace(/^www\./, '');

            return {
                title: d.title || d.publisher || d.siteName || '',
                description: compactDescription(d.description || ''),
                imageUrl: d.image?.url || '',
                favicon: d.logo?.url ||
                    `https://www.google.com/s2/favicons?sz=64&domain=${encodeURIComponent(domain)}`,
                domain
            };
        }

        function hasUsefulMetadata(data) {
            return Boolean(data && (data.title || data.description || data.imageUrl));
        }

        async function fetchWebData(url) {
            const cacheKey = `smart-link-meta:${url}`;

            // Cache 24 ชั่วโมง ลดการยิง API ซ้ำ
            try {
                const cached = JSON.parse(localStorage.getItem(cacheKey) || 'null');
                if (cached?.savedAt && Date.now() - cached.savedAt < 24 * 60 * 60 * 1000 && cached.data) {
                    return cached.data;
                }
            } catch {}

            let primary = null;

            // วิธีที่ 1: ดึง HTML ผ่าน proxy แล้วอ่าน OpenGraph/Twitter metadata
            try {
                primary = await fetchFromAllOrigins(url);
            } catch (error) {
                console.warn('AllOrigins metadata fetch failed:', error);
            }

            // ถ้าข้อมูลยังไม่ครบ ให้ Microlink ช่วยเติม Title / Description / Image
            if (!primary || !primary.title || !primary.description || !primary.imageUrl) {
                try {
                    const rich = await fetchFromMicrolink(url);
                    primary = {
                        ...(primary || {}),
                        title: primary?.title || rich.title,
                        description: primary?.description || rich.description,
                        imageUrl: primary?.imageUrl || rich.imageUrl,
                        favicon: primary?.favicon || rich.favicon,
                        domain: primary?.domain || rich.domain
                    };
                } catch (error) {
                    console.warn('Microlink metadata fetch failed:', error);
                }
            }

            if (!hasUsefulMetadata(primary)) return null;

            try {
                localStorage.setItem(cacheKey, JSON.stringify({
                    savedAt: Date.now(),
                    data: primary
                }));
            } catch {}

            return primary;
        }
